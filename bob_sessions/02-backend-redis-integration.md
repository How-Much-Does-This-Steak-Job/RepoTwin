```python
class RedisStore(Protocol):
    async def set_job_status(self, analysis_id: str, status: str) -> None
    async def get_job_status(self, analysis_id: str) -> Optional[str]
    async def set_progress(self, analysis_id: str, progress: int) -> None
    async def get_progress(self, analysis_id: str) -> Optional[int]
    async def set_result(self, analysis_id: str, result: dict) -> None
    async def get_result(self, analysis_id: str) -> Optional[dict]

# Factory selects implementation
def create_redis_store() -> RedisStore:
    if redis_available():
        return RedisStoreImpl()
    else:
        logger.warning("Redis unavailable, using in-memory store")
        return MemoryStore()
```

```python
class RedisStoreImpl:
    def __init__(self, redis_client: Redis):
        self.redis = redis_client
    
    async def set_job_status(
        self, 
        analysis_id: str, 
        status: str
    ) -> None:
        """Store job status with 1 hour TTL"""
        key = f"analysis:{analysis_id}:status"
        await self.redis.setex(key, 3600, status)
    
    async def get_job_status(
        self, 
        analysis_id: str
    ) -> Optional[str]:
        """Retrieve job status"""
        key = f"analysis:{analysis_id}:status"
        value = await self.redis.get(key)
        return value.decode() if value else None
    
    async def set_result(
        self, 
        analysis_id: str, 
        result: dict
    ) -> None:
        """Store analysis result with 24 hour TTL"""
        key = f"analysis:{analysis_id}:result"
        await self.redis.setex(
            key, 
            86400, 
            json.dumps(result)
        )
```

#### Memory Store Fallback (Generated by Bob)
```python
class MemoryStore:
    def __init__(self):
        self._storage: Dict[str, Any] = {}
        self._lock = asyncio.Lock()
    
    async def set_job_status(
        self, 
        analysis_id: str, 
        status: str
    ) -> None:
        """Store job status in memory"""
        async with self._lock:
            key = f"analysis:{analysis_id}:status"
            self._storage[key] = {
                "value": status,
                "expires_at": time.time() + 3600
            }
    
    async def get_job_status(
        self, 
        analysis_id: str
    ) -> Optional[str]:
        """Retrieve job status from memory"""
        async with self._lock:
            key = f"analysis:{analysis_id}:status"
            data = self._storage.get(key)
            
            if not data:
                return None
            
            # Check expiration
            if time.time() > data["expires_at"]:
                del self._storage[key]
                return None
            
            return data["value"]
```

```python
@router.post("/analyze", response_model=AnalyzeResponse)
async def create_analysis(
    request: AnalyzeRequest,
    store: RedisStore = Depends(get_redis_store)
) -> AnalyzeResponse:
    """Create new analysis job"""
    
    # Generate unique analysis ID
    analysis_id = f"analysis_{uuid.uuid4().hex[:12]}"
    
    # Store initial job status
    await store.set_job_status(analysis_id, "queued")
    await store.set_progress(analysis_id, 0)
    
    # Store job metadata
    metadata = {
        "repository": request.repositoryName,
        "change_request": request.changeRequest,
        "mode": request.mode,
        "created_at": datetime.utcnow().isoformat()
    }
    await store.set_metadata(analysis_id, metadata)
    
    # Queue analysis task (background)
    background_tasks.add_task(
        process_analysis,
        analysis_id,
        request,
        store
    )
    
    return AnalyzeResponse(
        analysisId=analysis_id,
        status="queued",
        estimatedDuration=120
    )

@router.get(
    "/analyses/{analysis_id}/status",
    response_model=StatusResponse
)
async def get_analysis_status(
    analysis_id: str,
    store: RedisStore = Depends(get_redis_store)
) -> StatusResponse:
    """Get analysis job status"""
    
    status = await store.get_job_status(analysis_id)
    
    if not status:
        raise HTTPException(
            status_code=404,
            detail="Analysis not found"
        )
    
    progress = await store.get_progress(analysis_id) or 0
    
    response = StatusResponse(
        analysisId=analysis_id,
        status=status,
        progress=progress,
        message=get_status_message(status, progress)
    )
    
    # Include result if completed
    if status == "completed":
        result = await store.get_result(analysis_id)
        if result:
            response.result = result
    
    return response
```
