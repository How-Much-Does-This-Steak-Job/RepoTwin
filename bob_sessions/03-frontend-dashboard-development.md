export default function ResultsPage() {
  const [shadowPR, setShadowPR] = useState<ShadowPR | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    loadAnalysisResults();
  }, []);

  const loadAnalysisResults = async () => {
    try {
      // Try API first, fallback to sample data
      const data = await apiClient.getAnalysisResult(analysisId);
      setShadowPR(data);
    } catch (error) {
      // Load sample data for demo
      const sampleData = await import('@/data/sample-shadow-pr.json');
      setShadowPR(sampleData.default);
    } finally {
      setLoading(false);
    }
  };

  if (loading) return <LoadingState />;
  if (!shadowPR) return <ErrorState />;

  return (
    <div className="min-h-screen bg-gray-950 text-gray-100">

          {/* Summary Cards */}

#### API Client (Generated by Bob)
```typescript
class APIClient {
  private baseURL: string;

  constructor() {
    this.baseURL = process.env.NEXT_PUBLIC_API_URL || 'http://localhost:8000';
  }

  async createAnalysis(request: AnalyzeRequest): Promise<AnalyzeResponse> {
    const response = await fetch(`${this.baseURL}/api/analyze`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(request)
    });

    if (!response.ok) {
      throw new Error('Failed to create analysis');
    }

    return response.json();
  }

  async getAnalysisStatus(analysisId: string): Promise<StatusResponse> {
    const response = await fetch(
      `${this.baseURL}/api/analyses/${analysisId}/status`
    );

    if (!response.ok) {
      throw new Error('Failed to get analysis status');
    }

    return response.json();
  }

  async pollUntilComplete(
    analysisId: string,
    onProgress?: (progress: number) => void
  ): Promise<ShadowPR> {
    while (true) {
      const status = await this.getAnalysisStatus(analysisId);

      if (onProgress) {
        onProgress(status.progress);
      }

      if (status.status === 'completed' && status.result) {
        return status.result;
      }

      if (status.status === 'failed') {
        throw new Error(status.error || 'Analysis failed');
      }

      // Wait 2 seconds before next poll
      await new Promise(resolve => setTimeout(resolve, 2000));
    }
  }
}

export const apiClient = new APIClient();
```
